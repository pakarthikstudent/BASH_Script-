# Task-16
# ------------
Write a shell script

Given Array => monitors=(vmstat iostat mpstat "sar -B")
	       ------------------------------------------
 -> iterate this array one by one - use for loop
 
 -> test  # case statement 
      -> vmstat => vmstat 2 3 - write vmstat status results to vmreport.log
      -> iostat => iostat 2 3 - write iostat status results to ioreport.log
      -> mpstat => mpstat 2 3 - write mpstat status results to process.log
      -> sar -B => filter last/recent 3 lines data to pages.log file
  |
 -> done 
-----------------------------------------------------------------------------
monitors=(vmstat iostat mpstat "sar -B")

n=2
m=3
for var in "${monitors[@]}"
do
  case $var in
    vmstat) $var $n $m >vmreport.log 2>&1 ;;
    iostat) $var $n $m >ioreport.log 2>&1 ;;
    mpstat) $var $n $m >process.log 2>&1 ;;
    "sar -B") $var|tail -n 3 >pages.log 2>&1 ;;
  esac
done
echo "------- DONE ----------"
